        function cameraErrorMessage(err) {
            const name = err && err.name ? err.name : '';
            const message = String((err && err.message) || err || '');

            if (name === 'NotFoundError' || name === 'DevicesNotFoundError' || /requested device not found|device not found|no camera/i.test(message)) {
                return 'Kamera tidak ditemukan. Di Windows, pastikan webcam terdeteksi di Pengaturan > Bluetooth & perangkat > Kamera dan tidak sedang dipakai aplikasi lain. Di Android, pastikan kamera berfungsi di aplikasi Kamera. Tutup aplikasi lain yang memakai kamera lalu muat ulang halaman.';
            }
            if (name === 'NotAllowedError' || name === 'PermissionDeniedError' || /permission denied|not allowed/i.test(message)) {
                return 'Izin kamera ditolak. Izinkan kamera untuk Chrome pada izin situs. Di Windows, buka Pengaturan > Privasi & keamanan > Kamera dan aktifkan akses kamera serta akses untuk aplikasi desktop. Di Android, buka Setelan > Aplikasi > Chrome > Izin > Kamera, lalu pilih Izinkan.';
            }
            if (name === 'NotReadableError' || name === 'TrackStartError' || /could not start video source|device in use/i.test(message)) {
                return 'Kamera terdeteksi tetapi tidak dapat dibuka. Tutup aplikasi lain yang menggunakan kamera (misalnya Camera, Teams, Zoom, WhatsApp) lalu coba lagi.';
            }
            if (name === 'OverconstrainedError' || name === 'ConstraintNotSatisfiedError') {
                return 'Pengaturan kamera tidak cocok dengan perangkat. Aplikasi akan mencoba kamera lain; muat ulang halaman lalu tekan Nyalakan Kamera kembali.';
            }
            if (name === 'SecurityError' || !window.isSecureContext || !navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
                return 'Browser tidak menyediakan akses kamera. Buka alamat Web App Google Apps Script yang diawali https:// di Chrome, bukan melalui pratinjau/editor atau halaman http://.';
            }
            return 'Kamera gagal dibuka. ' + (message || 'Pastikan kamera tersedia, izin Chrome aktif, dan coba lagi.');
        }

        async function collectCameraChoices() {
            const android = isAndroidDevice();
            const preferredFacing = android ? 'environment' : 'user';
            let probeStream = null;
            let preferredDeviceId = '';
            let probeError = null;

            // Minta akses kamera terlebih dahulu. Ini sekaligus mengungkap label
            // perangkat pada Chrome Android/Windows dan memilih kamera default.
            try {
                probeStream = await navigator.mediaDevices.getUserMedia({
                    video: {
                        facingMode: { ideal: preferredFacing },
                        width: { ideal: 1280 },
                        height: { ideal: 720 }
                    },
                    audio: false
                });
                const track = probeStream.getVideoTracks()[0];
                if (track && typeof track.getSettings === 'function') {
                    preferredDeviceId = track.getSettings().deviceId || '';
                }
            } catch (err) {
                probeError = err;
                // Tolak dengan pesan jelas bila pengguna/browser memblokir izin.
                if (err && (err.name === 'NotAllowedError' || err.name === 'PermissionDeniedError' || err.name === 'SecurityError')) {
                    throw err;
                }
                // Jika constraint kamera ideal ditolak, coba tanpa constraint.
                try {
                    probeStream = await navigator.mediaDevices.getUserMedia({ video: true, audio: false });
                    const track = probeStream.getVideoTracks()[0];
                    if (track && typeof track.getSettings === 'function') {
                        preferredDeviceId = track.getSettings().deviceId || '';
                    }
                } catch (fallbackErr) {
                    probeError = fallbackErr || probeError;
                }
            } finally {
                // Tutup stream uji sebelum html5-qrcode membuka stream utama.
                stopTemporaryCameraStream(probeStream);
            }

            let devices = [];
            try {
                devices = await navigator.mediaDevices.enumerateDevices();
            } catch (err) {
                console.warn('Daftar perangkat kamera tidak dapat dibaca.', err);
            }

            const cameras = devices.filter(d => d && d.kind === 'videoinput' && d.deviceId);
            const rearPattern = /back|rear|environment|belakang|camera2 0/i;
            const rearCameras = cameras.filter(c => rearPattern.test(c.label || ''));
            const ordered = android && rearCameras.length
                ? rearCameras.concat(cameras.filter(c => !rearPattern.test(c.label || '')))
                : cameras.slice();

            const choices = [];
            const addChoice = choice => {
                if (!choice) return;
                const key = typeof choice === 'string'
                    ? 'id:' + choice
                    : 'constraints:' + JSON.stringify(choice);
                if (!choices.some(item => item.key === key)) choices.push({ key, value: choice });
            };

            // Android: kamera belakang yang dikenali lebih dahulu; bila nama
            // kamera tersembunyi, gunakan kamera yang berhasil diuji. Windows:
            // gunakan webcam default yang berhasil diuji, lalu webcam lainnya.
            if (android && rearCameras.length) rearCameras.forEach(c => addChoice(c.deviceId));
            addChoice(preferredDeviceId);
            ordered.forEach(c => addChoice(c.deviceId));

            // Fallback berbasis constraint ideal tidak memaksa perangkat yang
            // tidak ada, jadi lebih ramah webcam Windows dan variasi Android.
            addChoice({ facingMode: { ideal: preferredFacing } });
            addChoice({});
            addChoice({ facingMode: { ideal: android ? 'user' : 'environment' } });

            return { choices: choices.map(item => item.value), probeError, cameraCount: cameras.length };
        }

        async function startScanner() {
            if (scannerStarting || scannerRunning) return;
            scannerStarting = true;
            setScannerButtons(false, true);

            if (typeof Html5Qrcode === 'undefined') {
                scannerStarting = false;
                setScannerButtons(false, false);
                alert('Modul pemindai QR belum berhasil dimuat. Periksa koneksi internet, muat ulang halaman, lalu coba lagi.');
                return;
            }

            if (!window.isSecureContext || !navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
                scannerStarting = false;
                setScannerButtons(false, false);
                alert(cameraErrorMessage(new Error('Akses kamera memerlukan HTTPS dan dukungan getUserMedia.')));
                return;
            }

            const config = {
                fps: 10,
                qrbox: (viewfinderWidth, viewfinderHeight) => {
                    const size = Math.max(160, Math.floor(Math.min(viewfinderWidth, viewfinderHeight) * 0.70));
                    return { width: Math.min(size, viewfinderWidth), height: Math.min(size, viewfinderHeight) };
                },
                aspectRatio: 1.0
            };
            let lastError = null;
            let started = false;

            try {
                const result = await collectCameraChoices();
                if (!result.choices.length && result.probeError) lastError = result.probeError;

                for (const cameraChoice of result.choices) {
                    let scanner = null;
                    try {
                        const reader = document.getElementById('reader');
                        if (reader) reader.innerHTML = '';
                        scanner = new Html5Qrcode('reader', { verbose: false });
                        await scanner.start(cameraChoice, config, onScanSuccess, onScanFailure);
                        html5QrcodeScanner = scanner;
                        scannerRunning = true;
                        started = true;
                        setScannerButtons(true, false);
                        break;
                    } catch (err) {
                        lastError = err;
                        console.warn('Gagal memakai pilihan kamera; mencoba pilihan berikutnya.', err);
                        if (scanner) {
                            try { await scanner.stop(); } catch (stopError) { /* kamera belum aktif */ }
                            try { scanner.clear(); } catch (clearError) { /* area dibersihkan pada percobaan berikutnya */ }
                        }
                    }
                }

                if (!started) throw (lastError || new Error('Tidak ada kamera yang dapat dibuka.'));
            } catch (err) {
                html5QrcodeScanner = null;
                scannerRunning = false;
                setScannerButtons(false, false);
                alert(cameraErrorMessage(err));
            } finally {
                scannerStarting = false;
                if (!scannerRunning) setScannerButtons(false, false);
            }
        }

        async function stopScanner() {
            const scanner = html5QrcodeScanner;
            if (!scanner) {
                scannerRunning = false;
                setScannerButtons(false, false);
                return;
            }

            try {
                if (scannerRunning) await scanner.stop();
            } catch (err) {
                console.warn('Kamera sudah berhenti atau tidak dapat dihentikan secara normal.', err);
            }
            try { scanner.clear(); } catch (err) { console.warn('Pembersihan area kamera:', err); }

            if (html5QrcodeScanner === scanner) html5QrcodeScanner = null;
            scannerRunning = false;
            setScannerButtons(false, false);
        }

        function processAttendance(nis, status, keterangan) {

                        displayAttendanceResult(response);
                    })
                    .recordAttendance(nis, status, keterangan);
            } else {
                // LocalStorage Mock Implementation
                const students = JSON.parse(localStorage.getItem('smpn2_students')) || SAMPLE_STUDENTS;
                const found = students.find(s => s.nis === nis);

                if (!found) {
                    displayAttendanceResult({
                        status: "error",
                        message: `NIS (${nis}) tidak terdaftar dalam sistem data siswa!`
                    });
                    return;
                }

                const todayStr = new Date().toLocaleDateString('id-ID');
                const timeStr = new Date().toLocaleTimeString('id-ID');
                let logs = JSON.parse(localStorage.getItem('smpn2_absen_logs') || '[]');

                const already = logs.find(l => l.nis === nis && l.tanggal === todayStr);
                if (already) {
                    displayAttendanceResult({
                        status: "warning",
                        message: `Siswa ${found.nama} (${found.kelas}) SUDAH absen hari ini jam ${already.waktu}`,
                        student: found
                    });
                    return;
                }

                const newLog = {
                    id: "ABS-" + Date.now(),
                    tanggal: todayStr,
                    waktu: timeStr,
                    nis: found.nis,
                    nama: found.nama,
                    kelas: found.kelas,
                    status: status,
                    keterangan: keterangan
                };

                logs.push(newLog);
                localStorage.setItem('smpn2_absen_logs', JSON.stringify(logs));

                displayAttendanceResult({
                    status: "success",
                    message: "Absensi Berhasil!",
                    student: found,
                    time: timeStr,
                    attendanceStatus: status
                });
            }
        }

        function displayAttendanceResult(res) {
            const card = document.getElementById('scanResultCard');
            const icon = document.getElementById('scanResultIcon');
            const badge = document.getElementById('scanResultBadge');
            const nama = document.getElementById('scanResultNama');
            const detail = document.getElementById('scanResultDetail');
            const time = document.getElementById('scanResultTime');

            card.classList.remove('hidden', 'bg-emerald-50', 'border-emerald-300', 'bg-amber-50', 'border-amber-300', 'bg-rose-50', 'border-rose-300');

            if (res.status === 'success') {
                card.classList.add('bg-emerald-50', 'border-emerald-300');
                icon.className = 'bg-emerald-500 text-white p-3 rounded-2xl shadow-md';
                icon.innerHTML = '<i class="fa-solid fa-circle-check text-2xl"></i>';
                
                badge.className = 'inline-block px-2.5 py-0.5 rounded-full text-xs font-bold bg-emerald-200 text-emerald-800 mb-1';
                badge.textContent = res.attendanceStatus || 'HADIR';
                
                nama.textContent = res.student.nama;
                detail.textContent = `NIS: ${res.student.nis} | Kelas: ${res.student.kelas}`;
                time.innerHTML = `<i class="fa-solid fa-clock mr-1"></i> Terekam Jam ${res.time} WIB`;
            } 
            else if (res.status === 'warning') {
                card.classList.add('bg-amber-50', 'border-amber-300');
                icon.className = 'bg-amber-500 text-white p-3 rounded-2xl shadow-md';
                icon.innerHTML = '<i class="fa-solid fa-triangle-exclamation text-2xl"></i>';
                
                badge.className = 'inline-block px-2.5 py-0.5 rounded-full text-xs font-bold bg-amber-200 text-amber-800 mb-1';
                badge.textContent = 'SUDAH ABSEN';
                
                nama.textContent = res.student ? res.student.nama : 'Peringatan';
                detail.textContent = res.message;
                time.textContent = '';
            } 
            else {
                card.classList.add('bg-rose-50', 'border-rose-300');
                icon.className = 'bg-rose-500 text-white p-3 rounded-2xl shadow-md';
                icon.innerHTML = '<i class="fa-solid fa-circle-xmark text-2xl"></i>';
                
                badge.className = 'inline-block px-2.5 py-0.5 rounded-full text-xs font-bold bg-rose-200 text-rose-800 mb-1';
                badge.textContent = 'GAGAL / TIDAK ADA';
                
                nama.textContent = 'Data Tidak Ditemukan';
                detail.textContent = res.message;
                time.textContent = '';
            }

            refreshDashboardData();
        }

        // --- REKAP ABSENSI FUNCTIONS ---
        function loadRekapData() {
            const filterTgl = document.getElementById('filterTanggal').value;
            const filterKls = document.getElementById('filterKelas').value;

            if (isGasEnvironment()) {
                google.script.run
                    .withSuccessHandler(logs => {
                        renderRekapTable(logs);
                    })
                    .getAttendanceLogs(filterTgl, filterKls);
            } else {
                let logs = JSON.parse(localStorage.getItem('smpn2_absen_logs') || '[]');
                const selectedDateStr = filterTgl ? new Date(filterTgl).toLocaleDateString('id-ID') : null;
                
                const filtered = logs.filter(log => {
                    const matchDate = !selectedDateStr || log.tanggal === selectedDateStr;
                    const matchKelas = filterKls === 'ALL' || log.kelas === filterKls;
                    return matchDate && matchKelas;
                });
                
                renderRekapTable(filtered);
            }
        }

        function renderRekapTable(logs) {
            const tbody = document.getElementById('rekapTableBody');
            document.getElementById('rekapCountText').textContent = `Menampilkan ${logs.length} Data Record Presensi`;

            if (logs.length === 0) {
                tbody.innerHTML = `<tr><td colspan="6" class="px-6 py-8 text-center text-slate-400">Tidak ada record absensi sesuai kriteria filter.</td></tr>`;
                return;
            }

            tbody.innerHTML = logs.map(l => `
                <tr class="hover:bg-slate-50 transition-colors">
                    <td class="px-6 py-3.5 text-xs text-slate-600 font-medium">${l.tanggal} <span class="text-slate-400 font-mono ml-1">(${l.waktu})</span></td>
                    <td class="px-6 py-3.5 font-mono text-xs font-bold text-blue-900">${l.nis}</td>
                    <td class="px-6 py-3.5 font-bold text-slate-900">${l.nama}</td>
                    <td class="px-6 py-3.5"><span class="px-2 py-1 rounded bg-slate-100 text-slate-700 text-xs font-bold">${l.kelas}</span></td>
                    <td class="px-6 py-3.5">${getStatusBadge(l.status)}</td>
                    <td class="px-6 py-3.5 text-xs text-slate-500">${l.keterangan || '-'}</td>
                </tr>
            `).join('');
        }

        function getStatusBadge(status) {
            switch((status || '').toLowerCase()) {
                case 'hadir':
                    return `<span class="px-2.5 py-1 rounded-full bg-emerald-100 text-emerald-800 text-xs font-bold border border-emerald-200">Hadir</span>`;
                case 'sakit':
                    return `<span class="px-2.5 py-1 rounded-full bg-amber-100 text-amber-800 text-xs font-bold border border-amber-200">Sakit</span>`;
                case 'izin':
                    return `<span class="px-2.5 py-1 rounded-full bg-indigo-100 text-indigo-800 text-xs font-bold border border-indigo-200">Izin</span>`;
                default:
                    return `<span class="px-2.5 py-1 rounded-full bg-rose-100 text-rose-800 text-xs font-bold border border-rose-200">Alpa</span>`;
            }
        }

                    .withSuccessHandler(res => {

                            closeModalAddSiswa();
                            fetchStudentsList();
                            refreshDashboardData();
                        }
                    })
                    .addStudent(studentData);
            } else {
                let students = JSON.parse(localStorage.getItem('smpn2_students')) || SAMPLE_STUDENTS;
                if (students.some(s => s.nis === studentData.nis)) {
                    alert(`NIS ${studentData.nis} sudah terdaftar!`);
                    return;
                }
                studentData.terdaftar = new Date().toLocaleDateString('id-ID');
                students.push(studentData);
                localStorage.setItem('smpn2_students', JSON.stringify(students));
                
                alert("Siswa berhasil ditambahkan!");
                closeModalAddSiswa();
                fetchStudentsList();
                refreshDashboardData();
            }
        }
